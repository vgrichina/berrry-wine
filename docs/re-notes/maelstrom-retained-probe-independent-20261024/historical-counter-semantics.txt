2617:   ;; Hit counters: set_count writes addr into slot N and zeros its count,
2618:   ;; bumping $hit_count_n so the run loop includes this slot. Caller passes
2619:   ;; slots 0..N-1 in order (contiguous); $hit_count_n = max(slot+1).
2620:   (func (export "set_count") (param $slot i32) (param $addr i32)
2621:     (local $base i32)
2622:     (local.set $base (i32.add (global.get $HIT_COUNT_BASE)
2623:                               (i32.shl (local.get $slot) (i32.const 3))))
2624:     ;; Arming the same address twice must not throw the count away. The slots
2625:     ;; live in linear memory, which worker instances share, and every spawn
2626:     ;; re-arms all of them (lib/thread-manager.js) -- so an app that started a
2627:     ;; thread mid-run reset every counter to zero and the flag reported 0 for
2628:     ;; addresses it had already counted hundreds of thousands of times. Use
2629:     ;; clear_counts to deliberately start over.
2630:     (if (i32.ne (i32.load (local.get $base)) (local.get $addr))
2631:       (then
2632:         (i32.store          (local.get $base) (local.get $addr))
2633:         (i32.store offset=4 (local.get $base) (i32.const 0))))
2634:     (if (i32.gt_s (i32.add (local.get $slot) (i32.const 1)) (global.get $hit_count_n))
2635:       (then (global.set $hit_count_n (i32.add (local.get $slot) (i32.const 1)))))
2636:     (call $dbg_recompute))
2637:   (func (export "get_count") (param $slot i32) (result i32)
2638:     (i32.load offset=4 (i32.add (global.get $HIT_COUNT_BASE)
2639:                                 (i32.shl (local.get $slot) (i32.const 3)))))
2640:   (func (export "clear_counts") (global.set $hit_count_n (i32.const 0)) (call $dbg_recompute))
