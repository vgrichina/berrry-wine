    };
    // Own only downs actually forwarded by this bridge. Repeated downs still
    // reach the renderer, but each physical key needs just one eventual up.
    // Keep the original VK/side metadata: modifiers can share a VK and the
    // browser may report a different VK after a layout/modifier change.
    const deliveredKeys = new Map();
    const physicalKey = e => e.code || `${e.keyCode}:${e.location | 0}`;
    const releaseDeliveredKeys = () => {
      const keys = Array.from(deliveredKeys.values());
      deliveredKeys.clear();
      altEnterSwallowed = false;
      for (const { vk, info } of keys) {
        renderer.handleKeyUp(vk, { ...info, repeat: false });
      }
    };
    // Element focus changes (including the iOS proxy) are ordinary routing,
    // not a loss of browser focus. Do not capture their non-bubbling blur.
    window.addEventListener('blur', releaseDeliveredKeys);
    document.addEventListener('visibilitychange', () => {
      if (document.hidden) releaseDeliveredKeys();
    });
    const handleKeyDown = (e) => {
      if (shouldIgnorePageKey(e)) return;
      if (e.keyCode === 13 && e.altKey && activeAltEnterFullscreen()) {
        e.preventDefault();
        e.stopPropagation();
        if (!e.repeat) toggleBrowserFullscreen(e);
        altEnterSwallowed = true;
        return;
      }
      const vk = e.keyCode;
      unlockRunningAudio();
      const info = {
        code: e.code || '', location: e.location | 0, repeat: !!e.repeat,
      };
      renderer.handleKeyDown(vk, info);
      const key = physicalKey(e);
      if (!deliveredKeys.has(key)) deliveredKeys.set(key, { vk, info });
      suppressTextInputEvents = 0;
      if (!keepForBrowser(e)) {
        const charCode = charCodeFromKeyEvent(e);
        if (charCode) {
          startInputProfile('keydown-char', e, { vk, charCode });
          renderer.handleKeyPress(charCode);
          renderer._profileMark && renderer._profileMark('browser-handler-end');
          suppressTextInputEvents = 2;
          e.preventDefault();
        } else if (!isPrintableKey(e)) {
          e.preventDefault();
        }
      }
    };
    const handleKeyUp = (e) => {
      if (e.keyCode === 13 && altEnterSwallowed) {
        altEnterSwallowed = false;
        e.preventDefault();
        return;
      }
      const key = physicalKey(e);
      const owned = deliveredKeys.get(key);
      if (!owned) return;
      // A down delivered before focus moved into a page control still owns
      // its up. Conversely, a stale/unowned up cannot release another input
      // producer's key after our blur/hidden cancellation.
      deliveredKeys.delete(key);
      renderer.handleKeyUp(owned.vk, {
        ...owned.info, repeat: !!e.repeat,
      });
    };
