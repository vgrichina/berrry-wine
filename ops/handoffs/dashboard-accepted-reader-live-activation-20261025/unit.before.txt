# /etc/systemd/system/wine-ops.service
[Unit]
Description=Wine Assembly dashboard (loopback only)
After=network.target

[Service]
Type=simple
User=user
WorkingDirectory=/home/user/wine-assembly
ExecStart=/usr/bin/env node ops/server.js --port=8098
Restart=on-failure
RestartSec=5
UMask=0077
NoNewPrivileges=true

[Install]
WantedBy=multi-user.target
